Escape special HTML characters // This is CRITICAL to prevent the input text from being interpreted as live HTML tags // and to safely display characters like '<' and '>' let htmlContent = rawText .replace(/&/g, "&") // Must be done first! .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); // 3. Replace new line characters with HTML
tags // The \n represents the new line character created when pressing 'Enter' in the textarea htmlContent = htmlContent.replace(/\n/g, "
"); // 4. Wrap the content in a

tag (optional, for block-level content) // You could also iterate and wrap paragraphs if you wanted. htmlContent = `

${htmlContent}

`; // 5. Output the final HTML to the output textarea document.getElementById('htmlOutput').value = htmlContent; }
